<!-- ============================================================
     PWA: مؤشر الأوفلاين + تسجيل الـ Service Worker
     ============================================================ -->
<script>
  /* ---------- (1) مؤشر الأوفلاين + شريط التنبيه ---------- */
  (function () {
    var bar  = document.getElementById('offlineBar');
    var chip = document.getElementById('offlineChip');

    /* نلزق الشريط تحت هيدر الصفحة الحالية (عشان يبان على كل الصفحات) */
    function placeBar() {
      if (!bar) return;
      var page = document.querySelector('.page.active');
      var header = page ? page.querySelector('header') : null;
      if (header && bar.previousElementSibling !== header) {
        header.insertAdjacentElement('afterend', bar);
      }
    }

    function sync() {
      var offline = (navigator.onLine === false);
      document.body.classList.toggle('is-offline', offline);
      if (bar)  bar.hidden  = !offline;
      if (chip) chip.hidden = !offline;
      placeBar();
    }

    window.addEventListener('online', sync);
    window.addEventListener('offline', sync);
    window.addEventListener('load', sync);
    document.addEventListener('visibilitychange', sync);

    /* كل ما صفحة تتفتح (switchPage) نعيد مكان الشريط */
    document.querySelectorAll('.page').forEach(function (p) {
      new MutationObserver(sync).observe(p, { attributes: true, attributeFilter: ['class'] });
    });

    sync();
  })();

  /* ---------- (2) تسجيل الـ Service Worker ---------- */
  if ('serviceWorker' in navigator) {
    window.addEventListener('load', function () {
      navigator.serviceWorker.register('service-worker.js').then(function (reg) {
        console.log('✅ Service Worker registered — scope:', reg.scope);

        /* بعد ما التطبيق يخلّص، نخزّن كل الأعلام أوفلاين (بالراحة في الخلفية) */
        function cacheFlagsOffline() {
          var list = (typeof countries !== 'undefined' && countries) ? countries : null;
          if (!list || !list.length) return;
          if (navigator.onLine === false) return;

          var urls = list
            .filter(function (c) { return c && c.code; })
            .map(function (c) { return flagUrl(c.code); });

          var target = navigator.serviceWorker.controller || reg.active;
          if (target && urls.length) {
            target.postMessage({ type: 'CACHE_URLS', urls: urls, cacheName: 'safr-flags-v1.0.0' });
            console.log('🚩 طلبنا تخزين ' + urls.length + ' علم أوفلاين');
          }
        }

        if ('requestIdleCallback' in window) {
          setTimeout(function () { requestIdleCallback(cacheFlagsOffline, { timeout: 4000 }); }, 3500);
        } else {
          setTimeout(cacheFlagsOffline, 3500);
        }
      }).catch(function (err) {
        console.warn('⚠️ فشل تسجيل الـ Service Worker:', err.message);
      });
    });
  }
</script>
